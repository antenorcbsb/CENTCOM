CANAIS RÁDIO (CORRIGIDO) async function carregarCanais() { try { const dados = await obterDadosAba(GIDS.canais); if (dados.length < 2) return; // Cabeçalhos das colunas (Ex: Canal BSB, Canal ROB, Canal REPC, etc.) const cabecalhos = dados[0]; const linhas = dados.slice(1); const container = document.getElementById('container-canais'); // Mapeamento de cores por tipo de rede/coluna const coresRede = { 'Canal BSB': '#38bdf8', // Ciano 'Canal ROB': '#fb923c', // Laranja 'Canal REPC': '#facc15', // Amarelo 'Canal SIRESP': '#4ade80',// Verde 'BSB_DMO': '#a855f7' // Roxo }; const eventosMap = {}; linhas.forEach(col => { const dataInicio = col[0]; const dataFim = col[1]; const evento = col[2] || 'Teatro Operacional'; if (estaAtivoNoIntervalo(dataInicio, dataFim)) { if (!eventosMap[evento]) { eventosMap[evento] = []; } // Percorre todas as colunas de canais (da coluna D/índice 3 até ao fim da tabela) for (let i = 3; i < col.length; i++) { const nomeCanal = col[i] ? col[i].trim() : ''; const nomeRede = cabecalhos[i] ? cabecalhos[i].trim() : `Canal ${i}`; if (nomeCanal.length > 0) { // Evita duplicados no mesmo evento const jaExiste = eventosMap[evento].some(item => item.canal === nomeCanal && item.rede === nomeRede); if (!jaExiste) { eventosMap[evento].push({ rede: nomeRede, canal: nomeCanal, cor: coresRede[nomeRede] || '#38bdf8' }); } } } } }); const eventosKeys = Object.keys(eventosMap); if (eventosKeys.length === 0) { container.innerHTML = '

Nenhum canal ativo para o dia de hoje.

'; return; } container.innerHTML = eventosKeys.map(evento => { const listaCanais = eventosMap[evento]; const badgesHtml = listaCanais.map(item => { return ` ${item.canal} `; }).join(''); return `
${ICONE_RADIO_TELECOM}

${evento}

${badgesHtml}
`; }).join(''); } catch (e) { console.error("Erro nos Canais:", e); } }
